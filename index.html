Conexão via ngrok WebSocket

📋 Configuração do Servidor

  1. Baixe e instale o ngrok: https://ngrok.com/download
  2. Crie uma conta no ngrok (grátis)
  3. Obtenha seu token authtoken
  4. Execute no terminal:
ngrok authtoken SEU_TOKEN_AQUI
ngrok tcp 8080
  • Copie a URL gerada (ex: tcp://0.tcp.ngrok.io:12345)
  • Execute o servidor WebSocket (código abaixo)
    • 📄 Código do Servidor WebSocket
    // 📄 server.js - Salve e execute com: node server.js
    const WebSocket = require('ws');
    const http = require('http');

    const server = http.createServer();
    const wss = new WebSocket.Server({ server });

    const rooms = new Map();

    wss.on('connection', (ws) => {
      let currentRoom = null;
      let userName = null;

      ws.on('message', (data) => {
        const msg = JSON.parse(data);
        
        switch(msg.type) {
          case 'join':
            currentRoom = msg.room;
            userName = msg.userName;
            if (!rooms.has(currentRoom)) {
              rooms.set(currentRoom, new Set());
            }
            rooms.get(currentRoom).add(ws);
            break;
          case 'message':
            if (currentRoom && rooms.has(currentRoom)) {
              const message = {
                ...msg,
                timestamp: new Date().toISOString(),
                userName: userName
              };
              rooms.get(currentRoom).forEach(client => {
                if (client !== ws && client.readyState === WebSocket.OPEN) {
                  client.send(JSON.stringify(message));
                }
              });
            }
            break;
        }
      });

      ws.on('close', () => {
        if (currentRoom && rooms.has(currentRoom)) {
          rooms.get(currentRoom).delete(ws);
        }
      });
    });

    server.listen(8080, () => {
      console.log('🚀 Servidor WebSocket rodando na porta 8080');
      console.log('📋 Execute: ngrok tcp 8080');
      console.log('🔗 Copie a URL ws://... e cole no campo acima');
    });
